=== Доступ к приложению ===

1. Добавьте запись в /etc/hosts:
   echo "$(minikube ip)  {{ .Values.ingress.host }}" | sudo tee -a /etc/hosts

2. Запустите Minikube tunnel (в отдельном терминале):
   minikube tunnel

3. Откройте в браузере:
   Frontend:  https://{{ .Values.ingress.host }}
   API:       https://{{ .Values.ingress.host }}/api/v1/
   Swagger:   https://{{ .Values.ingress.host }}/swagger/
{{- if .Values.flower.enabled }}
   Flower:    https://{{ .Values.ingress.host }}/flower
{{- end }}

=== Полезные команды ===

# Статус подов:
kubectl get pods -n {{ .Release.Namespace }}

# Логи backend:
kubectl logs -n {{ .Release.Namespace }} -l app.kubernetes.io/component=backend -f

{{- if (index .Values "celery-worker").enabled }}
# Логи celery-worker:
kubectl logs -n {{ .Release.Namespace }} -l app.kubernetes.io/component=celery-worker -f
{{- end }}

# Обновить релиз:
bash helm/scripts/deploy.sh

# Удалить релиз:
helm uninstall {{ .Release.Name }} -n {{ .Release.Namespace }}
